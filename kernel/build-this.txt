# 1. Go to the kernel source tree
cd ~/kernel-build/linux-6.12.1

# 2. Backup current config (optional but recommended)
cp .config .config.backup-$(date +%Y%m%d-%H%M%S)

# 3. Edit options in the config (interactive)
make menuconfig
#   - Change whatever you want
#   - Exit and save when done

# ---- BUILD STAGE ----

# 4. Rebuild the kernel and modules
make -j"$(nproc)"

# 5. Install modules for this kernel version (6.12.1)
/usr/bin/sudo make modules_install

# ---- INSTALL BOOTABLE KERNEL ----
# *** IMPORTANT: use the EFI-stub image, NOT vmlinux ***

# 6. Copy EFI-stub kernel image into /boot
/usr/bin/sudo cp arch/arm64/boot/vmlinuz.efi /boot/vmlinuz-6.12.1

# 7. Copy System.map and config so tools & initramfs match
/usr/bin/sudo cp System.map /boot/System.map-6.12.1
/usr/bin/sudo cp .config    /boot/config-6.12.1

# 8. (Safety) Make sure the broken raw image is not lying around
#    This avoids a second, non-working 6.12.1 entry in GRUB.
if [ -f /boot/vmlinux-6.12.1 ]; then
    /usr/bin/sudo rm /boot/vmlinux-6.12.1
fi

# ---- INITRAMFS & GRUB ----

# 9. Regenerate initramfs for 6.12.1 (update existing)
/usr/bin/sudo update-initramfs -u -k 6.12.1

# 10. Regenerate GRUB menu so it sees the new kernel image
/usr/bin/sudo update-grub

# 11. Reboot and select the "Ubuntu, with Linux 6.12.1" entry
/usr/bin/sudo reboot

# 12. After boot, verify the running kernel
uname -r
# should show: 6.12.1
